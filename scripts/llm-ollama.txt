Важно: после установки и запуска приложения надо разрешить обращение к облачным моделям!

# Проверить, установлена ли Ollama
ollama --version

# Посмотреть доступные команды
ollama --help

# Скачать модель
ollama pull gemma4:31b-cloud 

# Запустить модель и поговорить с ней прямо в терминале
ollama run gemma4:31b-cloud 

# Посмотреть установленные модели
ollama list

# Посмотреть модели, которые сейчас запущены
ollama ps

# Остановить запущенную модель
ollama stop qwen3:8b

# Удалить скачанную модель
ollama rm qwen3:8b

# Скопировать модель под другим именем
ollama cp qwen3:8b my-model
